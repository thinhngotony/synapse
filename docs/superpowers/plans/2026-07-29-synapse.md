# Synapse — Universal AI Tool Sync — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Replace skillshare with a single Go binary that syncs skills, agents, MCP configs, and plugins across omp, Copilot, Cursor, and OpenCode.

**Architecture:** CLI tool (`synapse`) with a YAML config at `~/.config/synapse/synapse.yaml` as single source of truth. Processors per resource type (skills, agents, MCP, plugins, rules) write to native tool paths via symlinks for files and MCP format adapters for JSON/TOML configs.

**Tech Stack:** Go 1.22+, Cobra (CLI), `gopkg.in/yaml.v3`, stdlib `encoding/json`

**Spec:** `docs/superpowers/specs/2026-07-29-synapse-design.md`

```go
// internal/config/config.go
package config

import (
    "os"
    "path/filepath"
    "gopkg.in/yaml.v3"
)

type Config struct {
    Version int              `yaml:"version"`
    Source  string           `yaml:"source"`
    GitRoot string           `yaml:"git_root"`
    Ignore  []string         `yaml:"ignore"`
    Audit   AuditConfig      `yaml:"audit"`
    Targets map[string]Target `yaml:"targets"`
    MCP     map[string]MCPServer `yaml:"mcp"`
    Plugins map[string]PluginSource `yaml:"plugins"`
}

type Target struct {
    Skills       string `yaml:"skills"`
    Agents       string `yaml:"agents"`
    MCP          string `yaml:"mcp"`
    Plugins      string `yaml:"plugins"`
    Settings     string `yaml:"settings,omitempty"`
    Rules        string `yaml:"rules,omitempty"`
    PluginSkills string `yaml:"plugin_skills,omitempty"`
}

type MCPServer struct {
    Command string            `yaml:"command"`
    Args    []string          `yaml:"args"`
    Env     map[string]string `yaml:"env,omitempty"`
    EnvFrom []string          `yaml:"envFrom,omitempty"`
}

type PluginSource struct {
    Repo    string   `yaml:"repo"`
    Targets []string `yaml:"targets"`
}

type AuditConfig struct {
    BlockThreshold string `yaml:"block_threshold"`
}

// Load reads synapse.yaml from ~/.config/synapse/
func Load() (*Config, error) {
    home, _ := os.UserHomeDir()
    path := filepath.Join(home, ".config", "synapse", "synapse.yaml")
    data, err := os.ReadFile(path)
    if err != nil {
        return nil, err
    }
    var cfg Config
    if err := yaml.Unmarshal(data, &cfg); err != nil {
        return nil, err
    }
    if cfg.Source == "" {
        cfg.Source = filepath.Join(home, ".config", "synapse")
    }
    // Expand ~ in paths
    cfg.expandPaths(home)
    return &cfg, nil
}

func (c *Config) expandPaths(home string) {
    expand := func(p *string) {
        if len(*p) >= 2 && (*p)[:2] == "~/" {
            *p = filepath.Join(home, (*p)[2:])
        }
    }
    expand(&c.Source)
    for k, t := range c.Targets {
        expand(&t.Skills)
        expand(&t.Agents)
        expand(&t.MCP)
        expand(&t.Plugins)
        expand(&t.Settings)
        expand(&t.Rules)
        expand(&t.PluginSkills)
        c.Targets[k] = t
    }
}
```

```go
// internal/sync/engine.go
package sync

import (
    "fmt"
    "github.com/thinhngotony/synapse/internal/config"
)

type Stats struct {
    SkillsCreated  int
    SkillsRemoved  int
    AgentsCreated  int
    AgentsRemoved  int
}

func Run(cfg *config.Config, targetFilter string) (*Stats, error) {
    stats := &Stats{}
    targets := cfg.Targets
    if targetFilter != "" {
        t, ok := targets[targetFilter]
        if !ok {
            return nil, fmt.Errorf("unknown target: %s", targetFilter)
        }
        targets = map[string]config.Target{targetFilter: t}
    }

    if err := syncSkills(cfg.Source, targets, stats); err != nil {
        return stats, fmt.Errorf("skills: %w", err)
    }
    if err := syncAgents(cfg.Source, targets, stats); err != nil {
        return stats, fmt.Errorf("agents: %w", err)
    }
    return stats, nil
}
```

```go
// internal/sync/skills.go
package sync

import (
    "fmt"
    "os"
    "path/filepath"
    "github.com/thinhngotony/synapse/internal/config"
)

func syncSkills(sourceDir string, targets map[string]config.Target, stats *Stats) error {
    skillsDir := filepath.Join(sourceDir, "skills")
    entries, err := os.ReadDir(skillsDir)
    if err != nil {
        return fmt.Errorf("read skills dir: %w", err)
    }

    for _, entry := range entries {
        if !entry.IsDir() {
            continue
        }
        skillName := entry.Name()
        srcPath := filepath.Join(skillsDir, skillName)

        for _, target := range targets {
            if target.Skills == "" {
                continue
            }
            dstDir := filepath.Join(target.Skills, skillName)
            if err := os.MkdirAll(filepath.Dir(dstDir), 0755); err != nil {
                return fmt.Errorf("mkdir %s: %w", filepath.Dir(dstDir), err)
            }

            // Walk source skill dir and symlink each file
            filepath.Walk(srcPath, func(srcFile string, info os.FileInfo, err error) error {
                if err != nil {
                    return err
                }
                rel, _ := filepath.Rel(skillName, filepath.Base(srcFile))
                if rel == "." {
                    return nil
                }
                dstFile := filepath.Join(dstDir, rel)

                // Remove existing
                os.Remove(dstFile)
                if err := os.Symlink(srcFile, dstFile); err != nil {
                    return fmt.Errorf("symlink %s → %s: %w", srcFile, dstFile, err)
                }
                stats.SkillsCreated++
                return nil
            })
        }
    }
    return nil
}
```

```go
package sync

import (
    "fmt"
    "os"
    "path/filepath"
    "github.com/thinhngotony/synapse/internal/config"
)

func syncAgents(sourceDir string, targets map[string]config.Target, stats *Stats) error {
    agentsDir := filepath.Join(sourceDir, "agents")
    entries, err := os.ReadDir(agentsDir)
    if err != nil {
        if os.IsNotExist(err) {
            return nil // no agents configured
        }
        return fmt.Errorf("read agents dir: %w", err)
    }

    for _, entry := range entries {
        if entry.IsDir() || filepath.Ext(entry.Name()) != ".md" {
            continue
        }
        srcPath := filepath.Join(agentsDir, entry.Name())

        for _, target := range targets {
            if target.Agents == "" {
                continue
            }
            dstPath := filepath.Join(target.Agents, entry.Name())
            os.MkdirAll(filepath.Dir(dstPath), 0755)
            os.Remove(dstPath)
            if err := os.Symlink(srcPath, dstPath); err != nil {
                return fmt.Errorf("symlink agent %s: %w", entry.Name(), err)
            }
            stats.AgentsCreated++
        }
    }
    return nil
}
```

```go
    "encoding/json"

    ConfigPath string

func (a *CopilotAdapter) Path() string { return a.ConfigPath }

type copilotConfig struct {
    MCPServers map[string]copilotServer `json:"mcpServers"`
}

    Type    string            `json:"type"`
    Command string            `json:"command"`
    Args    []string          `json:"args,omitempty"`
    Env     map[string]string `json:"env,omitempty"`
}

func (a *CopilotAdapter) Read() (*ToolConfig, error) {
    data, err := os.ReadFile(a.ConfigPath)
    if err != nil {
        if os.IsNotExist(err) {
            return &ToolConfig{
                Servers:    map[string]Server{},
                RawEntries: map[string]interface{}{},
            }, nil
        }
        return nil, err
    }
    var raw copilotConfig
    if err := json.Unmarshal(data, &raw); err != nil {
        return nil, fmt.Errorf("parse %s: %w", a.ConfigPath, err)
    }
    cfg := &ToolConfig{
        Servers:    make(map[string]Server),
        RawEntries: make(map[string]interface{}),
    }
    for name, s := range raw.MCPServers {
        cfg.Servers[name] = Server{
            Name:    name,
            Command: s.Command,
            Args:    s.Args,
            Env:     s.Env,
        }
    }
    return cfg, nil
}

func (a *CopilotAdapter) Write(cfg *ToolConfig, servers []Server) error {
    merged := make(map[string]copilotServer)
    // Preserve existing non-synapse servers
    for name, s := range cfg.Servers {
        merged[name] = copilotServer{
            Type:    "stdio",
            Command: s.Command,
            Args:    s.Args,
            Env:     s.Env,
        }
    }
    // Upsert synapse servers
    for _, s := range servers {
        merged[s.Name] = copilotServer{
            Type:    "stdio",
            Command: s.Command,
            Args:    s.Args,
            Env:     s.Env,
        }
    }
    output := copilotConfig{MCPServers: merged}
    data, err := json.MarshalIndent(output, "", "  ")
    if err != nil {
        return err
    }
    os.MkdirAll(filepath.Dir(a.ConfigPath), 0755)
    return os.WriteFile(a.ConfigPath, append(data, '\n'), 0644)
}
```

```go
package sync

import (
    "fmt"
    "os"

)

func syncMCP(cfg *config.Config, targets map[string]config.Target) error {
    if len(cfg.MCP) == 0 {
        return nil
    }

    // Build canonical server list from config
    var servers []mcp.Server
    for name, s := range cfg.MCP {
        // Resolve envFrom into env by reading from os.Getenv
        env := make(map[string]string)
        for k, v := range s.Env {
            env[k] = v
        }
        for _, key := range s.EnvFrom {
            if val := os.Getenv(key); val != "" {
                env[key] = val
            }
        }
        servers = append(servers, mcp.Server{
            Name:    name,
            Command: s.Command,
            Args:    s.Args,
            Env:     env,
        })
    }

    for name, target := range targets {
        if target.MCP == "" {
            continue
        }
        adapter := mcpAdapterFor(name, target.MCP)
        existing, err := adapter.Read()
        if err != nil {
            return fmt.Errorf("read MCP config for %s: %w", name, err)
        }
        if err := adapter.Write(existing, servers); err != nil {
            return fmt.Errorf("write MCP config for %s: %w", name, err)
        }
    }
    return nil
}

func mcpAdapterFor(targetName, path string) mcp.MCPTarget {
    switch targetName {
    case "opencode":
        return &mcp.OpenCodeAdapter{ConfigPath: path}
    case "cursor":
        return &mcp.CursorAdapter{ConfigPath: path}
    case "omp":
        return &mcp.ClaudeAdapter{CopilotAdapter: mcp.CopilotAdapter{ConfigPath: path}}
    default:
        // copilot + any unknown → copilot format
        return &mcp.CopilotAdapter{ConfigPath: path}
    }
}
```

```go
    "path/filepath"

type Meta struct {
    Name        string `json:"name"`
    Version     string `json:"version"`
    Marketplace string // derived from git remote or config

// Resolve reads plugin metadata from the source directory.
// Looks for .claude-plugin/plugin.json first, then plugin.json.
func Resolve(sourceDir string) (*Meta, error) {
    paths := []string{
        filepath.Join(sourceDir, ".claude-plugin", "plugin.json"),
        filepath.Join(sourceDir, "plugin.json"),
    }
    for _, p := range paths {
        data, err := os.ReadFile(p)
        if err != nil {
            continue
        }
        var m Meta
        if err := json.Unmarshal(data, &m); err != nil {
            continue
        }
        if m.Name != "" && m.Version != "" {
            return &m, nil
        }
    }
    return nil, fmt.Errorf("no valid plugin.json found in %s", sourceDir)
}
```

```go
    }
    out, err := json.MarshalIndent(m, "", "  ")
```

```go
// internal/sync/plugins.go
package sync

import (
    "fmt"
    "os"
    "path/filepath"
    "github.com/thinhngotony/synapse/internal/config"
    "github.com/thinhngotony/synapse/internal/plugin"
    "github.com/thinhngotony/synapse/internal/settings"
)

func syncPlugins(cfg *config.Config, targets map[string]config.Target) error {
    pluginsDir := filepath.Join(cfg.Source, "plugins")
    entries, err := os.ReadDir(pluginsDir)
    if err != nil {
        if os.IsNotExist(err) {
            return nil
        }
        return fmt.Errorf("read plugins dir: %w", err)
    }

    for _, entry := range entries {
        if !entry.IsDir() {
            continue
        }
        name := entry.Name()
        srcPath := filepath.Join(pluginsDir, name)

        // Resolve plugin metadata
        meta, err := plugin.Resolve(srcPath)
        if err != nil {
            return fmt.Errorf("resolve plugin %s: %w", name, err)
        }

        // Find which targets this plugin should sync to
        pluginCfg, ok := cfg.Plugins[name]
        if !ok {
            continue // configured but not in plugin list
        }

        for _, targetName := range pluginCfg.Targets {
            target, ok := targets[targetName]
            if !ok || target.Plugins == "" {
                continue
            }

            // Build target path: <pluginDir>/<marketplace>/<name>/<version>/
            marketplace := "superpowers-marketplace" // default
            // ponytail: hardcoded marketplace mapping, read from plugin.json if we add it later
            if name == "warp" {
                marketplace = "claude-code-warp"
            }
            if name == "understand-anything" {
                marketplace = "understand-anything"
            }

            dstPath := filepath.Join(target.Plugins, marketplace, meta.Name, meta.Version)
            os.MkdirAll(filepath.Dir(dstPath), 0755)
            os.RemoveAll(dstPath) // clean stale version
            if err := os.Symlink(srcPath, dstPath); err != nil {
                return fmt.Errorf("symlink plugin %s → %s: %w", srcPath, dstPath, err)
            }

            // Merge settings.json
            if target.Settings != "" {
                pluginID := fmt.Sprintf("%s@%s", meta.Name, marketplace)
                if err := settings.MergeEnabledPlugins(target.Settings, []string{pluginID}); err != nil {
                    return fmt.Errorf("merge settings for %s: %w", targetName, err)
                }
            }
        }
    }
    return nil
}
```

```go
package validate

import (
    "fmt"
    "os"
    "path/filepath"
    "github.com/thinhngotony/synapse/internal/config"
)

type Issue struct {
    Severity string // CRITICAL, WARNING
    Message  string
}

func Run(cfg *config.Config) []Issue {
    var issues []Issue

    // Check source dir exists
    if _, err := os.Stat(cfg.Source); err != nil {
        issues = append(issues, Issue{"CRITICAL", fmt.Sprintf("source dir does not exist: %s", cfg.Source)})
        return issues
    }

    // Check each skill has SKILL.md
    skillsDir := filepath.Join(cfg.Source, "skills")
    if entries, err := os.ReadDir(skillsDir); err == nil {
        for _, e := range entries {
            if !e.IsDir() { continue }
            skillMD := filepath.Join(skillsDir, e.Name(), "SKILL.md")
            if _, err := os.Stat(skillMD); os.IsNotExist(err) {
                issues = append(issues, Issue{"WARNING", fmt.Sprintf("skill %s missing SKILL.md", e.Name())})
            }
        }
    }

    // Check each plugin has plugin.json
    pluginsDir := filepath.Join(cfg.Source, "plugins")
    if entries, err := os.ReadDir(pluginsDir); err == nil {
        for _, e := range entries {
            if !e.IsDir() { continue }
            found := false
            for _, p := range []string{".claude-plugin/plugin.json", "plugin.json"} {
                if _, err := os.Stat(filepath.Join(pluginsDir, e.Name(), p)); err == nil {
                    found = true; break
                }
            }
            if !found {
                issues = append(issues, Issue{"CRITICAL", fmt.Sprintf("plugin %s missing plugin.json", e.Name())})
            }
        }
    }

    // Check MCP servers have resolvable commands
    for name, srv := range cfg.MCP {
        if _, err := exec.LookPath(srv.Command); err != nil {
            issues = append(issues, Issue{"WARNING", fmt.Sprintf("MCP server %s: command %s not in PATH", name, srv.Command)})
        }
    }

    // Check target paths are writable
    for name, t := range cfg.Targets {
        for _, p := range []string{t.Skills, t.Agents, t.MCP, t.Plugins} {
            if p == "" { continue }
            dir := filepath.Dir(p)
            if _, err := os.Stat(dir); os.IsNotExist(err) {
                issues = append(issues, Issue{"CRITICAL", fmt.Sprintf("target %s: directory does not exist: %s", name, dir)})
            }
        }
    }

    return issues
}
```

```go
                            }
                        }
                        return nil
```
```go
                    })
                }
            }

        },
```

### Task 6.1: Lockfile with stale-PID detection

- [ ] **Step 1: Implement lockfile**

Spec requirement (line 155): "Lockfile with stale-PID detection. Prevents concurrent sync; auto-clears dead PIDs." The lockfile lives at `~/.config/synapse/synapse.lock` and stores the PID of the running process. On acquire: if lockfile exists, read PID, check if process is alive via `os.FindProcess(pid).Signal(syscall.Signal(0))`. Dead → clear and proceed. Alive → error "another synapse sync is running (pid N)".

```go
// internal/sync/lockfile.go
package sync

import (
    "fmt"
    "os"
    "strconv"
    "syscall"
)

func acquireLock(lockPath string) (*os.File, error) {
    if existing, err := os.ReadFile(lockPath); err == nil {
        pid, _ := strconv.Atoi(string(existing))
        if proc, err := os.FindProcess(pid); err == nil {
            if proc.Signal(syscall.Signal(0)) == nil {
                return nil, fmt.Errorf("another synapse sync is running (pid %d)", pid)
            }
        }
        // pid is dead, clear stale lock
        os.Remove(lockPath)
    }
    f, err := os.Create(lockPath)
    if err != nil {
        return nil, err
    }
    fmt.Fprintf(f, "%d", os.Getpid())
    return f, nil
}

func releaseLock(f *os.File, lockPath string) {
    f.Close()
    os.Remove(lockPath)
}
```

```go
// internal/sync/rules.go
package sync

import (
    "fmt"
    "os"
    "path/filepath"
    "strings"
)

const (
    markerStart = "<!-- SYNAPSE_MANAGED_START -->"
    markerEnd   = "<!-- SYNAPSE_MANAGED_END -->"
)

func syncRules(cfg *config.Config, stats *Stats) error {
    rulesDir := filepath.Join(cfg.Source, "rules")
    entries, err := os.ReadDir(rulesDir)
    if err != nil {
        return nil // rules dir is optional
    }

    // Concatenate rules in alphabetical order
    var buf strings.Builder
    for _, e := range entries {
        if e.IsDir() || strings.HasPrefix(e.Name(), ".") {
            continue
        }
        content, err := os.ReadFile(filepath.Join(rulesDir, e.Name()))
        if err != nil {
            continue
        }
        fmt.Fprintf(&buf, "# Source: ~/.config/synapse/rules/%s\n\n%s\n\n", e.Name(), content)
    }
    managedBlock := buf.String()

    // Write to each target's rules_managed path
    for name, t := range cfg.Targets {
        if t.Rules == "" {
            continue
        }
        existing, _ := os.ReadFile(t.Rules)
        content := insertBetweenMarkers(string(existing), managedBlock)
        if err := os.WriteFile(t.Rules, []byte(content), 0644); err != nil {
            return fmt.Errorf("rules %s: %w", name, err)
        }
        stats.RulesSynced++
    }
    return nil
}

func insertBetweenMarkers(existing, newBlock string) string {
    startIdx := strings.Index(existing, markerStart)
    endIdx := strings.Index(existing, markerEnd)

    if startIdx == -1 || endIdx == -1 || endIdx <= startIdx {
        // Markers missing or malformed: regenerate entire file
        return fmt.Sprintf("%s\n%s\n%s\n", markerStart, newBlock, markerEnd)
    }

    // Preserve content before start and after end
    before := existing[:startIdx+len(markerStart)]
    after := existing[endIdx:]
    return before + "\n" + newBlock + after
}
```
